</style>

<body>

<h2>Factorial and Fibonacci</h2>

<input type="number" id="num" placeholder="Enter Number"> <button onclick="show()">Show</button>

<div id="result"></div>
<script>

function factorial(n)
{
let fact = 1;

for (let i = 1; i <= n; i++) {
fact = fact * i;

}
return fact;
}
function fibonacci(n) {
let a = 0;
let b = 1;
let series = "";

for (let i = 1; i <= n; i++) {
series += a + " ";

let c = a + b;
a = b;
b = c;
}
return series;
}
function show() {

let n = Number(document.getElementById("num").value); if (n < 0) { document.getElementById("result").innerHTML = "Please enter a positive number.";

return;
} document.getElementById("result").innerHTML = "Factorial = " + factorial(n) + "<br><br>" +

"Fibonacci Series = " + fibonacci(n); }

</script>
